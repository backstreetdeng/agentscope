    function stopExecutionNarrative() {
      if (executionTimer) clearInterval(executionTimer);
      executionTimer = null;
    }

    function startExecutionNarrative(agent, question) {
      stopExecutionNarrative();
      const startedAt = Date.now();
      const plan = buildExecutionPlan(agent, question);
      let cursor = 0;
      let lastHeartbeat = 0;

      appendLog("audit-note: plan/reasoning/quality 是页面阶段提示；tool-call 才是后端真实工具审计");

      executionTimer = setInterval(() => {
        const elapsed = Date.now() - startedAt;
        while (cursor < plan.length && elapsed >= plan[cursor].after) {
          const item = plan[cursor];
          if (item.stage) updateStep(item.stage[0], item.stage[1], item.stage[2]);
          if (item.flow) setFlow(item.flow);
          appendLog(item.log);
          cursor += 1;
        }
        if (elapsed > 36000 && elapsed - lastHeartbeat > 10000) {
          lastHeartbeat = elapsed;
          appendLog(`heartbeat: 模型仍在推理或等待工具结果，已耗时 ${Math.round(elapsed / 1000)}s`);
        }
      }, 900);
    }

    function escapeAttr(text) {
      return text.replaceAll("&", "&amp;").replaceAll('"', "&quot;").replaceAll("<", "&lt;").replaceAll(">", "&gt;");
    }

    function markdownToHtml(markdown) {
      const safe = (markdown || "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;");

      const lines = safe.split(/\r?\n/);
      let html = "";
      let inTable = false;
      for (const line of lines) {
        if (/^\|.+\|$/.test(line.trim())) {
          const cells = line.trim().slice(1, -1).split("|").map(c => c.trim());
          if (cells.every(c => /^:?-{3,}:?$/.test(c))) continue;
          if (!inTable) { html += "<table>"; inTable = true; }
          html += "<tr>" + cells.map(c => `<td>${formatInline(c)}</td>`).join("") + "</tr>";
          continue;
        }
        if (inTable) { html += "</table>"; inTable = false; }
        if (/^###\s+/.test(line)) html += `<h3>${formatInline(line.replace(/^###\s+/, ""))}</h3>`;
        else if (/^##\s+/.test(line)) html += `<h2>${formatInline(line.replace(/^##\s+/, ""))}</h2>`;
        else if (/^#\s+/.test(line)) html += `<h1>${formatInline(line.replace(/^#\s+/, ""))}</h1>`;
        else if (/^[-*]\s+/.test(line)) html += `<p>• ${formatInline(line.replace(/^[-*]\s+/, ""))}</p>`;
        else if (line.trim()) html += `<p>${formatInline(line)}</p>`;
      }
      if (inTable) html += "</table>";
      return html || `<div class="empty"><div><strong>没有返回内容</strong><p>请调整问题后重试。</p></div></div>`;
    }

    function formatInline(text) {
      return text.replace(/\*\*(.*?)\*\*/g, "<strong>$1</strong>");
    }

    async function runAudioWorkflow(animate = true) {
      updateAudioWorkflowStatus("run", "编排执行中");
      $("runAudioWorkflowBtn").disabled = true;
      try {
        const res = await fetch("/api/audio-tags/run", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            instruction: $("audioWorkflowInput").value.trim(),
            selected_tags: audioTagState.selectedTags
          })
        });
        const data = await res.json();
        if (!res.ok || data.error) throw new Error(data.error || `HTTP ${res.status}`);
        renderAudioTagWorkbench(data, animate);
        updateAudioWorkflowStatus("ok", `完成：${data.filtered_count}/${data.all_record_count} 条样本命中`);
        $("tagWorkbenchView").dataset.loaded = "true";
      } catch (err) {
        updateAudioWorkflowStatus("bad", "执行失败");
        $("taggedRecordPanel").innerHTML = `<div class="tag-empty-note">执行失败：${escapeAttr(err.message)}</div>`;
      } finally {
        $("runAudioWorkflowBtn").disabled = false;
      }
    }

    function exportTagCsv() {
      const records = audioTagState.records || [];
      if (!records.length) {
        alert("当前没有可导出的标签结果。");
        return;
      }
      const header = ["user_id", "audio_name", "region", "city_tier", "labels", "emotion_score", "intensity_score", "config_mapping", "evidence"];
      const rows = records.map(record => header.map(key => {
        const value = Array.isArray(record[key]) ? record[key].join("|") : record[key];
        return `"${String(value || "").replaceAll('"', '""')}"`;
      }).join(","));
      downloadText(`用户标签分析结果-${new Date().toISOString().slice(0, 10)}.csv`, [header.join(","), ...rows].join("\n"));
    }

    function switchMainView(view, activeNav = null) {
      const targetMap = {
        reports: "reportQueryView",
        tagging: "tagWorkbenchView",
        analysis: "analysisView"
      };
      const target = targetMap[view] || "analysisView";
      document.querySelectorAll(".workspace-view").forEach(item => {
        item.hidden = item.id !== target;
      });
      document.querySelectorAll(".nav-item").forEach(item => item.classList.remove("active"));
      (activeNav || document.querySelector(`.nav-item[data-view="${view}"]`))?.classList.add("active");
      if (view === "reports" && !$("reportList").dataset.loaded) {
        loadReportLibrary();
      }
      if (view === "tagging" && !$("tagWorkbenchView").dataset.loaded) {
        runAudioWorkflow();
      }
    }

    function formatFileSize(bytes) {
      if (!Number.isFinite(bytes)) return "-";
      if (bytes < 1024) return `${bytes} B`;
      if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
      return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
    }

    async function loadReportLibrary(queryValue = null) {
      const query = queryValue === null ? $("reportSearchInput").value.trim() : queryValue;
      if (queryValue !== null) $("reportSearchInput").value = queryValue;
      $("reportList").dataset.loaded = "true";
      $("reportLibraryMeta").textContent = "正在检索报告资产...";
      $("reportList").innerHTML = `<div class="empty"><div><strong>正在查询</strong><p>正在扫描 outputs/ppt 报告目录。</p></div></div>`;
      try {
        const res = await fetch(`/api/reports?q=${encodeURIComponent(query)}`);
        const data = await res.json();
        if (!res.ok || data.error) throw new Error(data.error || `HTTP ${res.status}`);
        renderReportLibrary(data);
      } catch (err) {
        $("reportLibraryMeta").textContent = "报告查询失败";
        $("reportList").innerHTML = `<div class="empty"><div><strong>查询失败</strong><p>${escapeAttr(err.message)}</p></div></div>`;
      }
    }

    function renderReportLibrary(data) {
      const items = data.items || [];
      const query = data.query || "";
      $("reportLibraryMeta").textContent = query
        ? `关键词“${query}”命中 ${items.length} 份报告，已按最近修改时间排序。`
        : `共发现 ${items.length} 份可用报告，已按最近修改时间排序。`;
      if (!items.length) {
        $("reportList").innerHTML = `<div class="empty"><div><strong>没有找到匹配报告</strong><p>可以换成车型、价格带、区域、用户圈层等关键词再试。</p></div></div>`;
        return;
      }
      $("reportList").innerHTML = items.map(item => {
        const openText = item.ext === ".html" ? "打开PPT" : "打开报告";
        return `
          <article class="report-item">
            <div class="report-item-head">
              <div class="report-name">${escapeAttr(item.name)}</div>
              <span class="report-type">${escapeAttr(item.type)}</span>
            </div>
            <div class="report-snippet">${escapeAttr(item.snippet || "暂无正文摘要")}</div>
            <div class="report-file-meta">
              <span>修改时间：${escapeAttr(item.modified)}</span>
              <span>大小：${formatFileSize(item.size)}</span>
              <span>路径：${escapeAttr(item.relative_path)}</span>
            </div>
            <div class="report-actions">
              <a class="link-btn primary" href="${escapeAttr(item.url)}" target="_blank" rel="noreferrer">${openText}</a>
              <a class="link-btn" href="${escapeAttr(item.url)}" download>下载文件</a>
            </div>
          </article>
        `;
      }).join("");
    }

    async function runAnalysis(agent = activeAgent, question = $("questionInput").value.trim()) {
      if (running) return;
      if (!question) { alert("请输入分析问题"); return; }
      setAgent(agent);
      running = true;
      $("runBtn").disabled = true;
      $("runBtn").textContent = "分析中";
      $(`state-${agent}`).textContent = "运行中";
      setStatus("run", "智能体运行中");
      $("eventLog").textContent = "开始接收智能体执行日志...";
      $("reportArea").innerHTML = `<div class="empty"><div><strong>智能体正在分析</strong><p>请观察左侧阶段进度。</p></div></div>`;
      $("mQuality").textContent = "-";
      $("mLength").textContent = "-";
      initSteps();
      startDemoProgress();

      updateStep("stage1", "done", "已识别业务场景与目标智能体。");
      updateStep("stage2", "running", "正在准备模型、工具与业务上下文。");
      appendLog(`Agent=${agent}`);
      startExecutionNarrative(agent, question);

      try {
        const res = await fetch("/api/chat", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ agent, question, analysis_type: $("analysisType").value })
        });
        const data = await res.json();
        if (!res.ok || data.error) throw new Error(data.error || `HTTP ${res.status}`);
        stopExecutionNarrative();

        updateStep("stage2", "done", "工具链准备完成。");
        setFlow(58);
        updateStep("stage3", "done", "已按业务框架完成推理。");
        setFlow(76);
        updateStep("stage4", "done", "已检查报告章节完整性。");
        setFlow(90);
        updateStep("stage5", "done", "完整报告已返回。");
        appendLog("complete: report returned");
        if (Array.isArray(data.tool_events) && data.tool_events.length) {
          appendLog(`audit: 后端返回真实工具审计 ${data.tool_events.length} 条`);
          data.tool_events.forEach(item => appendLog(`tool-call: ${item.name} / ${item.status} / ${item.detail}`));
        } else {
          appendLog("audit: 后端本轮未捕获到真实工具调用");
        }

        lastReport = data.answer || "";
        lastReportMeta = {
          agent,
          question,
          analysisType: $("analysisType").value || "自动识别",
          createdAt: new Date().toISOString()
        };
        $("reportArea").innerHTML = markdownToHtml(lastReport);
        $("mQuality").textContent = "通过";
        animateMetric("mLength", lastReport.length);
        finishDemoProgress(true);
        setStatus("ok", "分析完成");
      } catch (err) {
        updateStep("stage2", "done", "请求已发送。");
        updateStep("stage3", "running", "执行失败，请查看错误。");
        stopExecutionNarrative();
        appendLog(`ERROR: ${err.message}`);
        $("reportArea").innerHTML = `<div class="empty"><div><strong>分析失败</strong><p>${err.message}</p></div></div>`;
        finishDemoProgress(false);
        setStatus("bad", "分析失败");
      } finally {
        running = false;
        stopExecutionNarrative();
        $("runBtn").disabled = false;
        $("runBtn").textContent = "开始分析";
        $(`state-${agent}`).textContent = "待命";
      }
    }
